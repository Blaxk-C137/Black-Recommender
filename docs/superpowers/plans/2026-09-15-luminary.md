# Luminary Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build Luminary, a guided multi-media discovery app (movies, anime, manga, books) with a 4-step flow ending in a results page with Best Rated / Trending / AI Pick / browse sections.

**Architecture:** Next.js 15 App Router with URL params as the single source of truth for flow state. Server-side route handlers (`/api/tmdb`, `/api/anilist`, `/api/books`, `/api/ai/recommend`) wrap three external APIs and normalize everything to a universal `MediaItem`. TanStack Query on the client; Zustand only for modal state.

**Tech Stack:** Next.js 15, React 19, Tailwind v4, shadcn/ui, Zustand 5, TanStack Query 5, Framer Motion 12, `@google/genai`, Vitest.

**Spec:** `docs/superpowers/specs/2026-09-15-luminary-design.md` — the spec supersedes any conflicting detail in CLAUDE.md. Read both.

## Global Constraints

- 4 media types only: `movie`, `anime`, `manga`, `book`. No "animation" type.
- Anime + manga data: AniList GraphQL only (no Jikan, no MangaDex).
- TMDB genre sections MUST use `/discover/{movie|tv}` with `vote_count.gte=300` when sorting by `vote_average.desc`. Never genre-filter `top_rated` endpoints.
- 3 API keys only: `TMDB_API_KEY`, `GOOGLE_BOOKS_API_KEY`, `GEMINI_API_KEY`.
- URL params are the single source of truth for flow state. Zustand holds only `selectedItem` (modal state).
- subType vocabulary (URL `sub` param uses these exact strings): movie → `movie|series`; anime → `tv|movie|ova`; manga → `manga|manhwa|manhua`; book → `fiction|nonfiction`.
- Books have NO "Best Rated" and NO "Trending" sections — only "In Your Genres" (relevance) and "New Releases" (newest).
- Gemini failure hides the AI section silently; it never breaks the page.
- Stack: `npx shadcn@latest` (not shadcn-ui), `@google/genai` (not @google/generative-ai).
- Git commits: attribute solely to the user. NEVER add `Co-Authored-By: Claude` or `Generated with Claude Code` lines (per user's global CLAUDE.md — this overrides defaults).
- Every UI task ends with a dev-server walkthrough; every lib task ends with `npx vitest run <file>` passing.
- Working directory for all commands: `/home/blaxk/Downloads/.Downloads/scripts/blaxk_rec` (call it `$ROOT`).

- [ ] **Step 5: Write `lib/utils/formatters.ts`**

```ts
export function normalizeRating(score: number | null | undefined, scale: 10 | 100 = 10): number {
  if (score == null) return 0
  const value = scale === 100 ? score / 10 : score
  return Math.round(value * 10) / 10
}

export function stripHtml(html: string | null | undefined): string {
  if (!html) return ""
  return html.replace(/<[^>]*>/g, "").trim()
}

export function truncate(text: string, max = 200): string {
  if (text.length <= max) return text
  return text.slice(0, max).replace(/\s+\S*$/, "") + "…"
}

export function parseGenresParam(param: string | null | undefined): string[] {
  if (!param) return []
  return param
    .split(",")
    .map(g => g.trim())
    .filter(Boolean)
}
```

```ts
const IMG_BASE = "https://image.tmdb.org/t/p/w500"

export type TmdbKind = "movie" | "tv"

export const TMDB_GENRE_MAP: Record<string, number> = {
  Action: 28,
  Adventure: 12,
  Animation: 16,
  Comedy: 35,
  Crime: 80,
  Documentary: 99,
  Drama: 18,
  Family: 10751,
  Fantasy: 14,
  Historical: 36,
  Horror: 27,
  Music: 10402,
  Mystery: 9648,
  Reality: 10764,
  Romance: 10749,
  "Sci-Fi": 878,
  Thriller: 53,
  War: 10752,
  Western: 37,
}

const ID_TO_GENRE = Object.fromEntries(
  Object.entries(TMDB_GENRE_MAP).map(([name, id]) => [id, name])

export interface TmdbResult {
  id: number
  title?: string
  name?: string
  overview?: string
  poster_path?: string | null
  backdrop_path?: string | null
  vote_average?: number
  release_date?: string
  first_air_date?: string
  genre_ids?: number[]
  genres?: { id: number; name: string }[]
}

async function tmdbFetch<T>(
  path: string,
  params: Record<string, string | number> = {}
): Promise<T> {
  const key = process.env.TMDB_API_KEY
  if (!key) throw new Error("TMDB_API_KEY is not set")
  const url = new URL(`${BASE_URL}${path}`)
  for (const [k, v] of Object.entries(params)) url.searchParams.set(k, String(v))
  url.searchParams.set("api_key", key)
  const res = await fetch(url)
  if (!res.ok) throw new Error(`TMDB request failed: ${res.status} ${path}`)
  return (await res.json()) as T
}

export function mapTmdbItem(r: TmdbResult, kind: TmdbKind): MediaItem {
  const date = r.release_date ?? r.first_air_date ?? ""
  const genres = r.genres
    ? r.genres.map(g => g.name)
    : (r.genre_ids ?? []).map(id => ID_TO_GENRE[id]).filter(Boolean)
  return {
    id: r.id,
    title: r.title ?? r.name ?? "Untitled",
    description: r.overview ?? "",
    posterUrl: r.poster_path ? `${IMG_BASE}${r.poster_path}` : "",
    backdropUrl: r.backdrop_path
      ? `https://image.tmdb.org/t/p/w780${r.backdrop_path}`
      : undefined,
    rating: Math.round((r.vote_average ?? 0) * 10) / 10,
    year: date.slice(0, 4),
    genres,
    mediaType: "movie",
    subType: kind,
    source: "tmdb",
    externalId: String(r.id),
  }
}

export async function getBestRated(
  kind: TmdbKind,
  genreIds: number[],
  page = 1
): Promise<MediaItem[]> {
  const data = await tmdbFetch<{ results: TmdbResult[] }>(`/discover/${kind}`, {
    with_genres: genreIds.join(","),
    sort_by: "vote_average.desc",
    "vote_count.gte": 300,
    include_adult: false,
    page,
  })
  return data.results.map(r => mapTmdbItem(r, kind))
}

export async function getMoreByGenre(
  kind: TmdbKind,
  genreIds: number[],
  page = 1
): Promise<MediaItem[]> {
  const data = await tmdbFetch<{ results: TmdbResult[] }>(`/discover/${kind}`, {
    with_genres: genreIds.join(","),
    sort_by: "popularity.desc",
    include_adult: false,
    page,
  })
  return data.results.map(r => mapTmdbItem(r, kind))
}

export async function getTrending(kind: TmdbKind): Promise<MediaItem[]> {
  const data = await tmdbFetch<{ results: TmdbResult[] }>(`/trending/${kind}/week`)
  return data.results.map(r => mapTmdbItem(r, kind))
}

export async function getTmdbDetails(kind: TmdbKind, id: number): Promise<MediaItem> {
  const r = await tmdbFetch<TmdbResult>(`/${kind}/${id}`)
  return mapTmdbItem(r, kind)
}

export async function getTrailerUrl(kind: TmdbKind, id: number): Promise<string | null> {
  const data = await tmdbFetch<{ results: { key: string; type: string; site: string }[] }>(
    `/${kind}/${id}/videos`
  )
  const trailer = data.results.find(
    v => v.site === "YouTube" && (v.type === "Trailer" || v.type === "Teaser")
  )
  return trailer ? `https://www.youtube.com/embed/${trailer.key}` : null
}

export async function searchTmdb(query: string, kind: TmdbKind): Promise<MediaItem[]> {
  const data = await tmdbFetch<{ results: TmdbResult[] }>(`/search/${kind}`, {
    query,
    include_adult: false,
  })
  return data.results.filter(r => r.poster_path).map(r => mapTmdbItem(r, kind))
}
```

```ts
import type { MediaItem } from "@/lib/types/media.types"
import { stripHtml } from "@/lib/utils/formatters"

const ANILIST_URL = process.env.ANILIST_BASE_URL ?? "https://graphql.anilist.co"

export type AniListType = "ANIME" | "MANGA"
export type AniListSort = "SCORE_DESC" | "TRENDING_DESC" | "POPULARITY_DESC"

/** Selected-genre strings that are AniList TAGS (not genres) → tag names. */
const ANILIST_TAG_MAP: Record<string, string> = {
  Shonen: "Shounen",
  Shojo: "Shoujo",
  Seinen: "Seinen",
  Isekai: "Isekai",
  "School Life": "School",
  "Martial Arts": "Martial Arts",
}

export function splitAnilistGenres(genres: string[]): { genres: string[]; tags: string[] } {
  const tags: string[] = []
  const rest = genres.filter(g => {
    const tag = ANILIST_TAG_MAP[g]
    if (tag) {
      tags.push(tag)
      return false
    }
    return true
  })
  return { genres: rest, tags }
}

const FIELDS = `
  id
  title { romaji english }
  description(asHtml: false)
  averageScore
  coverImage { large }
  bannerImage
  startDate { year }
  genres
  format
  status
  episodes
  chapters
  siteUrl
`

const LIST_QUERY = `
query ($page: Int, $perPage: Int, $type: MediaType, $sort: [MediaSort], $genres: [String], $tags: [String], $countryOfOrigin: CountryCode, $formats: [MediaFormat], $search: String) {
  Page(page: $page, perPage: $perPage) {
    media(type: $type, sort: $sort, genre_in: $genres, tag_in: $tags, countryOfOrigin: $countryOfOrigin, format_in: $formats, search: $search, isAdult: false) {
      ${FIELDS}
    }
  }
}`

const DETAIL_QUERY = `
query ($id: Int) {
  Media(id: $id) {
    ${FIELDS}
  }
}`

export interface AniListMedia {
  id: number
  title: { romaji: string | null; english: string | null }
  description: string | null
  averageScore: number | null
  coverImage: { large: string | null } | null
  bannerImage: string | null
  startDate: { year: number | null } | null
  genres: string[] | null
  format: string | null
  status: string | null
  episodes: number | null
  chapters: number | null
  siteUrl: string | null
}

async function anilistRequest<T>(query: string, variables: Record<string, unknown>): Promise<T> {
  const res = await fetch(ANILIST_URL, {
    method: "POST",
    headers: { "Content-Type": "application/json", Accept: "application/json" },
    body: JSON.stringify({ query, variables }),
  })
  if (!res.ok) throw new Error(`AniList request failed: ${res.status}`)
  const json = (await res.json()) as { data?: T; errors?: { message: string }[] }
  if (json.errors?.length) throw new Error(`AniList error: ${json.errors[0].message}`)
  if (!json.data) throw new Error("AniList returned no data")
  return json.data
}

export function mapAnilistItem(m: AniListMedia, type: AniListType): MediaItem {
  return {
    id: m.id,
    title: m.title.english ?? m.title.romaji ?? "Untitled",
    description: stripHtml(m.description),
    posterUrl: m.coverImage?.large ?? "",
    backdropUrl: m.bannerImage ?? undefined,
    rating: m.averageScore ? Math.round(m.averageScore) / 10 : 0,
    year: m.startDate?.year ? String(m.startDate.year) : "",
    genres: m.genres ?? [],
    mediaType: type === "ANIME" ? "anime" : "manga",
    source: "anilist",
    externalId: String(m.id),
    externalUrl: m.siteUrl ?? undefined,
    status: m.status ?? undefined,
    episodeCount: m.episodes ?? undefined,
    chapterCount: m.chapters ?? undefined,
  }
}

export interface AniListQueryOpts {
  type: AniListType
  genres?: string[]
  countryOfOrigin?: "JP" | "KR" | "CN"
  formats?: string[]
  search?: string
  page?: number
}

export async function anilistMedia(sort: AniListSort, opts: AniListQueryOpts): Promise<MediaItem[]> {
  const { genres, tags } = splitAnilistGenres(opts.genres ?? [])
  const data = await anilistRequest<{ Page: { media: AniListMedia[] } }>(LIST_QUERY, {
    page: opts.page ?? 1,
    perPage: 20,
    type: opts.type,
    sort: [sort],
    genres: genres.length ? genres : undefined,
    tags: tags.length ? tags : undefined,
    countryOfOrigin: opts.countryOfOrigin,
    formats: opts.formats,
    search: opts.search,
  })
  return data.Page.media.map(m => mapAnilistItem(m, opts.type))
}

export async function getAniListDetails(id: number, type: AniListType): Promise<MediaItem> {
  const data = await anilistRequest<{ Media: AniListMedia }>(DETAIL_QUERY, { id })
  return mapAnilistItem(data.Media, type)
}

export async function searchAniList(query: string, type: AniListType): Promise<MediaItem[]> {
  return anilistMedia("POPULARITY_DESC", { type, search: query })
}
```

```ts
const BASE_URL = process.env.GOOGLE_BOOKS_BASE_URL ?? "https://www.googleapis.com/books/v1"

const GOOGLE_BOOKS_MAP: Record<string, string> = {
  Fantasy: "fantasy",
  "Sci-Fi": "science fiction",
  Romance: "romance",
  Mystery: "mystery",
  Thriller: "thriller",
  Horror: "horror",
  Historical: "historical fiction",
  Literary: "literary fiction",
  "Young Adult": "young adult fiction",
  Classic: "classics",
  "Self-Help": "self-help",
  Biography: "biography",
  History: "history",
  Science: "science",
  Psychology: "psychology",
  Philosophy: "philosophy",
  "True Crime": "true crime",
  Travel: "travel",
  Business: "business",
}

export type BookOrderBy = "relevance" | "newest"

export function buildBooksQuery(genres: string[]): string {
  return genres.map(g => `subject:${GOOGLE_BOOKS_MAP[g] ?? g}`).join(" ")
}

export interface GoogleVolume {
  id: string
  volumeInfo: {
    title?: string
    authors?: string[]
    description?: string
    averageRating?: number
    imageLinks?: { thumbnail?: string }
    publishedDate?: string
    categories?: string[]
    industryIdentifiers?: { type: string; identifier: string }[]
    pageCount?: number
  }
}

export function mapBookItem(v: GoogleVolume): MediaItem {
  const info = v.volumeInfo
  const thumbnail = info.imageLinks?.thumbnail?.replace("http://", "https://")
  const isbn13 = info.industryIdentifiers?.find(i => i.type === "ISBN_13")?.identifier
  return {
    id: v.id,
    title: info.title ?? "Untitled",
    description: info.description ?? "",
    posterUrl: thumbnail ?? (isbn13 ? `https://covers.openlibrary.org/b/isbn/${isbn13}-L.jpg` : ""),
    rating: Math.round((info.averageRating ?? 0) * 2 * 10) / 10,
    year: (info.publishedDate ?? "").slice(0, 4),
    genres: info.categories ?? [],
    mediaType: "book",
    source: "googlebooks",
    externalId: v.id,
    author: info.authors?.[0],
    pageCount: info.pageCount,
  }
}

export async function getBooks(
  genres: string[],
  orderBy: BookOrderBy,
  page = 0
): Promise<MediaItem[]> {
  const key = process.env.GOOGLE_BOOKS_API_KEY
  if (!key) throw new Error("GOOGLE_BOOKS_API_KEY is not set")
  const params = new URLSearchParams({
    q: buildBooksQuery(genres),
    printType: "books",
    maxResults: "20",
    orderBy,
    startIndex: String(page * 20),
    key,
  })
  const res = await fetch(`${BASE_URL}/volumes?${params}`)
  if (!res.ok) throw new Error(`Google Books request failed: ${res.status}`)
  const json = (await res.json()) as { items?: GoogleVolume[] }
  return (json.items ?? []).map(mapBookItem).filter(b => b.posterUrl)
}

export async function searchBooks(query: string): Promise<MediaItem[]> {
  const key = process.env.GOOGLE_BOOKS_API_KEY
  if (!key) throw new Error("GOOGLE_BOOKS_API_KEY is not set")
  const params = new URLSearchParams({ q: query, maxResults: "5", key })
  const res = await fetch(`${BASE_URL}/volumes?${params}`)
  if (!res.ok) throw new Error(`Google Books request failed: ${res.status}`)
  const json = (await res.json()) as { items?: GoogleVolume[] }
  return (json.items ?? []).map(mapBookItem).filter(b => b.posterUrl)
}
```

```ts
export async function GET(request: Request) {
  const { searchParams } = new URL(request.url)
  const section = searchParams.get("section") ?? ""
  const kind = (searchParams.get("kind") === "tv" ? "tv" : "movie") as TmdbKind
  const genres = parseGenresParam(searchParams.get("genres"))

  const page = Number(searchParams.get("page") ?? 1) || 1
  const id = Number(searchParams.get("id"))

  try {
    switch (section) {
      case "best-rated":
        return NextResponse.json(await getBestRated(kind, genreIds, page))
      case "trending":
        return NextResponse.json(await getTrending(kind))
      case "more":

      case "details":

      case "trailer":
        return NextResponse.json({ url: await getTrailerUrl(kind, id) })
      default:
        return NextResponse.json({ error: `Unknown section: ${section}` }, { status: 400 })
```

```ts
import type { MediaType } from "./constants"

export type SectionId = "best-rated" | "trending" | "in-genres" | "new-releases" | "more"

export const SECTION_TITLES: Record<SectionId, string> = {
  "best-rated": "★ Best Rated",
  trending: "🔥 Trending Now",
  "in-genres": "📚 In Your Genres",
  "new-releases": "🆕 New Releases",
  more: "🎯 More in Your Genres",
}

export function sectionsFor(type: MediaType): SectionId[] {
  return type === "book"
    ? ["in-genres", "new-releases", "more"]
    : ["best-rated", "trending", "more"]
}

export function buildSectionUrl(
  type: MediaType,
  sub: string,
  genres: string[],
  section: SectionId,
  page = 1
): string {
  const g = encodeURIComponent(genres.join(","))
  if (type === "book") {
    if (section === "new-releases")
      return `/api/books?section=newest&sub=${sub}&genres=${g}`
    if (section === "in-genres")
      return `/api/books?section=relevance&sub=${sub}&genres=${g}`
    return `/api/books?section=relevance&sub=${sub}&genres=${g}&page=${page}`
  }
  if (type === "anime" || type === "manga") {
    const t = type === "anime" ? "ANIME" : "MANGA"
    if (section === "best-rated")
      return `/api/anilist?section=best&type=${t}&sub=${sub}&genres=${g}`
    if (section === "trending")
      return `/api/anilist?section=trending&type=${t}&sub=${sub}&genres=${g}`
    return `/api/anilist?section=more&type=${t}&sub=${sub}&genres=${g}&page=${page}`
  }
  const kind = sub === "series" ? "tv" : "movie"
  if (section === "best-rated")
    return `/api/tmdb?section=best-rated&kind=${kind}&genres=${g}`
  if (section === "trending") return `/api/tmdb?section=trending&kind=${kind}`
  return `/api/tmdb?section=more&kind=${kind}&genres=${g}&page=${page}`
}
```

```ts
"use client"

import { useQuery, type UseQueryResult } from "@tanstack/react-query"
import { useSearchParams } from "next/navigation"
import type { MediaItem } from "@/lib/types/media.types"
import { parseGenresParam } from "@/lib/utils/formatters"
import type { MediaType } from "@/lib/utils/constants"
import { buildSectionUrl, type SectionId } from "@/lib/utils/sections"

const HOUR = 1000 * 60 * 60
const DAY = HOUR * 24

export function useSection(section: SectionId, page = 1): UseQueryResult<MediaItem[]> {
  const params = useSearchParams()
  const type = (params.get("type") ?? "movie") as MediaType
  const sub = params.get("sub") ?? ""
  const genres = parseGenresParam(params.get("genres"))
  const url = buildSectionUrl(type, sub, genres, section, page)

  return useQuery({
    queryKey: ["section", type, sub, genres, section, page],
    queryFn: async () => {
      const res = await fetch(url)
      if (!res.ok) {
        const body = (await res.json().catch(() => null)) as { error?: string } | null
        throw new Error(body?.error ?? `Request failed: ${res.status}`)
      }
      return (await res.json()) as MediaItem[]
    },
    staleTime: section === "trending" || section === "new-releases" ? HOUR : DAY,
  })
}
```

```ts
import { GoogleGenAI } from "@google/genai"

export interface AIRecommendation {
  title: string
  year: string
  reason: string
  hype: string
  searchQuery: string
}

export function stripCodeFences(text: string): string {
  return text.replace(/```(?:json)?\s*/g, "").replace(/```/g, "").trim()
}

export async function getAIRecommendation(
  mediaType: string,
  subType: string,
  genres: string[]
): Promise<AIRecommendation> {
  const key = process.env.GEMINI_API_KEY
  if (!key) throw new Error("GEMINI_API_KEY is not set")

  const prompt = `You are a media recommendation expert.

The user wants: ${mediaType} (${subType})
Their favorite genres: ${genres.join(", ")}

Recommend exactly ONE perfect title for them.
Respond in this exact JSON format:
{
  "title": "Title Name",
  "year": "2019",
  "reason": "One sentence why this matches their taste",
  "hype": "One punchy exciting sentence to make them want it",
  "searchQuery": "title name year"
}

Only respond with the JSON. No extra text.`

  const ai = new GoogleGenAI({ apiKey: key })
  const response = await ai.models.generateContent({
    model: process.env.GEMINI_MODEL ?? "gemini-2.5-flash",
    contents: prompt,
  })
  const text = response.text ?? ""
  try {
    return JSON.parse(stripCodeFences(text)) as AIRecommendation
  } catch {
    throw new Error(`Gemini returned unparseable output: ${text.slice(0, 200)}`)
  }
}
```

```ts
import { NextResponse } from "next/server"
import { searchAniList } from "@/lib/api/anilist"
import { searchBooks } from "@/lib/api/books"
import { getAIRecommendation } from "@/lib/api/gemini"
import { searchTmdb } from "@/lib/api/tmdb"
import type { MediaItem } from "@/lib/types/media.types"
import { parseGenresParam } from "@/lib/utils/formatters"

export async function GET(request: Request) {
  const { searchParams } = new URL(request.url)
  const type = searchParams.get("type") ?? "movie"
  const sub = searchParams.get("sub") ?? ""
  const genres = parseGenresParam(searchParams.get("genres"))

  try {
    const recommendation = await getAIRecommendation(type, sub, genres)

    // Best-effort poster lookup — never fails the request.
    let item: MediaItem | null = null
    try {
      if (type === "movie") {
        const kind = sub === "series" ? "tv" : "movie"
        item = (await searchTmdb(recommendation.searchQuery, kind))[0] ?? null
      } else if (type === "anime") {
        item = (await searchAniList(recommendation.searchQuery, "ANIME"))[0] ?? null
      } else if (type === "manga") {
        item = (await searchAniList(recommendation.searchQuery, "MANGA"))[0] ?? null
      } else {
        item = (await searchBooks(recommendation.searchQuery))[0] ?? null
      }
    } catch {
      item = null
    }

    return NextResponse.json({ recommendation, item })
  } catch (e) {
    return NextResponse.json(
      { error: e instanceof Error ? e.message : "Gemini error" },
      { status: 502 }
    )
  }
}
```

```ts
import type { MediaItem } from "@/lib/types/media.types"

const KEY = "luminary:list"

export function getList(): MediaItem[] {
  try {
    const raw = localStorage.getItem(KEY)
    if (!raw) return []
    const parsed = JSON.parse(raw)
    return Array.isArray(parsed) ? (parsed as MediaItem[]) : []
  } catch {
    return []
  }
}

export function isSaved(id: string | number): boolean {
  return getList().some(item => String(item.id) === String(id))
}

export function toggleSaved(item: MediaItem): boolean {
  const list = getList()
  const exists = list.some(i => String(i.id) === String(item.id))
  const next = exists ? list.filter(i => String(i.id) !== String(item.id)) : [...list, item]
  try {
    localStorage.setItem(KEY, JSON.stringify(next))
  } catch {
    // private mode / quota — treat as no-op
  }
  return !exists
}
```

```tsx
export function DetailModal() {
  const item = useUIStore(s => s.selectedItem)
  const close = () => useUIStore.getState().setSelectedItem(null)
  const params = useSearchParams()
  const [saved, setSaved] = useState(false)

  useEffect(() => {
    if (item) setSaved(isSaved(item.id))
  }, [item])

  // Extra details for TMDB/AniList sources (books already carry everything).
  const details = useQuery<MediaItem | null>({
    queryKey: ["details", item?.source, item?.externalId],
    queryFn: async () => {
      if (!item) return null
      if (item.source === "tmdb") {
        const kind = item.subType === "tv" ? "tv" : "movie"
        const res = await fetch(`/api/tmdb?section=details&kind=${kind}&id=${item.externalId}`)
        if (!res.ok) return null
        return (await res.json()) as MediaItem
      }
      if (item.source === "anilist") {
        const type = item.mediaType === "manga" ? "MANGA" : "ANIME"
        const res = await fetch(`/api/anilist?section=details&type=${type}&id=${item.externalId}`)
        if (!res.ok) return null
        return (await res.json()) as MediaItem
      }
      return null
    },
    enabled: !!item,
    staleTime: 1000 * 60 * 60 * 12,
  })

  // Trailer (TMDB only).
  const trailer = useQuery<{ url: string | null }>({
    queryKey: ["trailer", item?.source, item?.externalId],
    queryFn: async () => {
      if (!item || item.source !== "tmdb") return { url: null }
      const kind = item.subType === "tv" ? "tv" : "movie"
      const res = await fetch(`/api/tmdb?section=trailer&kind=${kind}&id=${item.externalId}`)
      if (!res.ok) return { url: null }
      return (await res.json()) as { url: string | null }
    },
    enabled: !!item && item.source === "tmdb",
    staleTime: 1000 * 60 * 60 * 12,
  })

  // Similar: same source, first genre.
  const similar = useQuery<MediaItem[]>({
    queryKey: ["similar", item?.source, item?.externalId, item?.genres[0]],
    queryFn: async () => {
      if (!item?.genres.length) return []
      const sub = params.get("sub") ?? ""
      const type = item.mediaType
      let url: string
      if (type === "movie") {
        const kind = item.subType === "tv" ? "tv" : "movie"
        url = `/api/tmdb?section=more&kind=${kind}&genres=${encodeURIComponent(item.genres[0])}`
      } else if (type === "anime" || type === "manga") {
        const t = type === "anime" ? "ANIME" : "MANGA"
        url = `/api/anilist?section=more&type=${t}&sub=${sub}&genres=${encodeURIComponent(item.genres[0])}`
      } else {
        url = `/api/books?section=relevance&sub=${sub}&genres=${encodeURIComponent(item.genres[0])}`
      }
      const res = await fetch(url)
      if (!res.ok) return []
      const all = (await res.json()) as MediaItem[]
      return all.filter(i => String(i.id) !== String(item.id)).slice(0, 8)
    },
    enabled: !!item,
    staleTime: 1000 * 60 * 60,
  })

  if (!item) return null

  const full = details.data
  const externalUrl =
    item.externalUrl ??
    (item.source === "tmdb"
      ? `https://www.themoviedb.org/${item.subType === "tv" ? "tv" : "movie"}/${item.externalId}`
      : undefined)

  return (
    <Dialog open onOpenChange={open => !open && close()}>
      <DialogContent className="max-h-[90dvh] overflow-y-auto p-0 max-w-3xl">
        <DialogTitle className="sr-only">{item.title}</DialogTitle>
        <button
          onClick={close}
          aria-label="Close"
          className="absolute right-3 top-3 z-10 rounded-full bg-black/60 p-2"
        >
          <X className="size-4" />
        </button>

        {full?.backdropUrl || item.backdropUrl ? (
          // eslint-disable-next-line @next/next/no-img-element
          <img
            src={(full?.backdropUrl ?? item.backdropUrl) as string}
            alt=""
            className="h-48 w-full object-cover md:h-64"
          />
        ) : (
          <div className="h-24 w-full bg-gradient-to-br from-card to-surface" />
        )}

        <div className="flex flex-col gap-4 p-6 md:flex-row">
          {item.posterUrl && (
            // eslint-disable-next-line @next/next/no-img-element
            <img
              src={item.posterUrl}
              alt={item.title}
              className="h-48 w-32 shrink-0 rounded-lg border border-border object-cover"
            />
          )}
          <div className="min-w-0">
            <h2 className="font-display text-2xl font-bold">{item.title}</h2>
            <p className="mt-1 font-mono text-sm text-muted-foreground">
              {[
                item.rating > 0 ? `★ ${item.rating.toFixed(1)}` : null,
                full?.year || item.year,
                ...(full?.genres ?? item.genres).slice(0, 3),
                item.author,
                item.episodeCount ? `${item.episodeCount} eps` : null,
                item.chapterCount ? `${item.chapterCount} ch` : null,
                item.pageCount ? `${item.pageCount} pages` : null,
                item.status,
              ]
                .filter(Boolean)
                .join(" • ")}
            </p>
            <p className="mt-3 text-sm leading-relaxed text-muted-foreground">
              {truncate(full?.description || item.description, 300)}

            <div className="mt-4 flex flex-wrap gap-2">
```
